## 一、存储层：Dexie.js (IndexedDB)

所有数据存在浏览器 IndexedDB 中，使用 Dexie.js 封装，提供类型安全的 TypeScript 接口。

数据库名：`flashmind`，当前版本：`1`

## 二、数据表定义

### 2.1 Deck（牌组）

```typescript
interface Deck {
  id: string              // UUID，主键
  name: string            // 牌组名，最长 50 字
  description?: string    // 描述，可选
  emoji?: string          // 封面 emoji，如 "📚"
  color: string           // 封面颜色 hex，如 "#6366F1"
  is_archived: boolean    // 是否归档
  created_at: number      // UTC 毫秒时间戳
  updated_at: number
}
```

**索引**：`id`（主键）、`is_archived`、`created_at`

---

### 2.2 Card（卡片）

```typescript
interface Card {
  id: string
  deck_id: string

  // 内容
  front: CardContent
  back: CardContent

  // FSRS-5 状态（直接映射 ts-fsrs 的 Card 结构）
  due: number             // 下次复习时间，UTC 毫秒
  stability: number       // 记忆稳定性
  difficulty: number      // 卡片难度
  elapsed_days: number    // 距上次复习天数
  scheduled_days: number  // 本次计划间隔天数
  reps: number            // 总复习次数
  lapses: number          // 遗忘次数
  state: CardState        // 学习状态
  last_review?: number    // 上次复习时间，UTC 毫秒

  // 元数据
  tags: string[]
  is_suspended: boolean
  created_at: number
  updated_at: number
}

interface CardContent {
  type: 'text' | 'markdown'
  text: string
  image_id?: string       // 关联图片 ID
}

enum CardState {
  New = 0,        // 新卡，从未复习
  Learning = 1,   // 学习中（间隔 < 1 天）
  Review = 2,     // 复习中（间隔 >= 1 天）
  Relearning = 3  // 重学（遗忘后重学）
}
```

**索引**：`id`、`deck_id`、`due`、`state`、`[deck_id+due]`（复合索引）

### 2.3 ReviewLog（复习日志）

每次打分后写入一条，用于统计和遗忘曲线分析。

```typescript
interface ReviewLog {
  id: string
  card_id: string
  deck_id: string         // 冗余存储，方便统计

  rating: Rating          // 用户打分（1-4）
  state: CardState        // 复习前状态
  due: number             // 复习前的计划到期时间
  stability: number
  difficulty: number
  elapsed_days: number
  scheduled_days: number
  review_duration_ms: number  // 从翻卡到打分的耗时

  reviewed_at: number     // UTC 毫秒
}

enum Rating {
  Again = 1,  // 忘了
  Hard  = 2,  // 模糊
  Good  = 3,  // 记得
  Easy  = 4   // 很熟
}
```

**索引**：`id`、`card_id`、`deck_id`、`reviewed_at`、`[deck_id+reviewed_at]`

## 三、Dexie Schema 声明

```typescript
// src/db/schema.ts
import Dexie, { type Table } from 'dexie'
import type { Deck, Card, ReviewLog, Media, Settings } from '../types'

class FlashMindDB extends Dexie {
  decks!: Table<Deck>
  cards!: Table<Card>
  reviewLogs!: Table<ReviewLog>
  media!: Table<Media>
  settings!: Table<Settings>

  constructor() {
    super('flashmind')
    this.version(1).stores({
      decks:      'id, is_archived, created_at',
      cards:      'id, deck_id, due, state, [deck_id+due]',
      reviewLogs: 'id, card_id, deck_id, reviewed_at, [deck_id+reviewed_at]',
      media:      'id',
      settings:   'id',
    })
  }
}

export const db = new FlashMindDB()
```

## 四、核心查询示例

### 4.1 获取今日复习队列

```typescript
// 指定牌组的到期卡片（复合索引，高效）
async function getDueCards(deckId: string, limit: number): Promise<Card[]> {
  return db.cards
    .where('[deck_id+due]')
    .between([deckId, 0], [deckId, Date.now()])
    .and(card => !card.is_suspended)
    .limit(limit)
    .toArray()
}

// 全部牌组的新卡
async function getNewCards(limit: number): Promise<Card[]> {
  return db.cards
    .where('state').equals(CardState.New)
    .and(card => !card.is_suspended)
    .limit(limit)
    .toArray()
}
```

### 4.2 复习后原子更新

```typescript
async function applyReview(
  card: Card,
  rating: Rating,
  durationMs: number
): Promise<void> {
  const f = fsrs()
  const now = new Date()
  const result = f.repeat(toFSRSCard(card), now)
  const scheduled = result[rating].card

  await db.transaction('rw', db.cards, db.reviewLogs, async () => {
    await db.cards.update(card.id, {
      due:            scheduled.due.getTime(),
      stability:      scheduled.stability,
      difficulty:     scheduled.difficulty,
      elapsed_days:   scheduled.elapsed_days,
      scheduled_days: scheduled.scheduled_days,
      reps:           scheduled.reps,
      lapses:         scheduled.lapses,
      state:          scheduled.state as unknown as CardState,
      last_review:    now.getTime(),
      updated_at:     now.getTime(),
    })
    await db.reviewLogs.add({
      id: crypto.randomUUID(),
      card_id:           card.id,
      deck_id:           card.deck_id,
      rating,
      state:             card.state,
      due:               card.due,
      stability:         card.stability,
      difficulty:        card.difficulty,
      elapsed_days:      card.elapsed_days,
      scheduled_days:    card.scheduled_days,
      review_duration_ms: durationMs,
      reviewed_at:       now.getTime(),
    })
  })
}
```

### 4.3 统计查询

```typescript
// 今日复习总数
async function getTodayCount(): Promise<number> {
  const startOfDay = new Date()
  startOfDay.setHours(0, 0, 0, 0)
  return db.reviewLogs
    .where('reviewed_at').aboveOrEqual(startOfDay.getTime())
    .count()
}

// 牌组状态分布
async function getDeckStats(deckId: string) {
  const cards = await db.cards.where('deck_id').equals(deckId).toArray()
  const now = Date.now()
  return {
    total:    cards.length,
    new:      cards.filter(c => c.state === CardState.New).length,
    learning: cards.filter(c => c.state === CardState.Learning).length,
    review:   cards.filter(c => c.state === CardState.Review).length,
    due:      cards.filter(c => c.due <= now && c.state !== CardState.New).length,
  }
}
```

## 五、数据迁移策略

schema 变化时通过 Dexie 版本升级处理：

```typescript
// V1 → V2：新增 tags 索引
this.version(2).stores({
  cards: 'id, deck_id, due, state, *tags, [deck_id+due]',
}).upgrade(tx =>
  tx.table('cards').toCollection().modify(card => {
    if (!card.tags) card.tags = []
  })
)
```

原则：**升级只加不减**，旧字段保留，新字段给默认值，保证向后兼容。

### 6.1 完整导出（JSON）

```json
{
  "version": 1,
  "exported_at": 1724800000000,
  "decks": [{ "id": "...", "name": "英语词汇", "color": "#6366F1" }],

    "due": 1724800000000, "stability": 4.5, "state": 2
  }],
  "review_logs": []
}
```

### 6.2 CSV 批量导入

```csv
front,back,tags
ephemeral,短暂的,英语;CET-6
ubiquitous,无处不在的,英语
```

- `tags` 列可选，多标签用 `;` 分隔
- 导入后所有卡片初始状态为 `CardState.New`
- 重复检测：front 完全相同则跳过并提示用户

## 七、存储容量估算

| 场景 | 单条大小 | 1万张卡片总量 |
|-----|---------|------------|
| 纯文字卡片 | ~300 bytes | ~3 MB |
| 复习日志（每卡10次） | ~150 bytes | ~15 MB |
| 含图片卡片（压缩后100KB/张） | ~100 KB | ~1 GB（1万张全图） |

**结论**：纯文字场景无需担心容量。图片较多时提示用户清理，每张图片压缩至 ≤200KB 上传。
